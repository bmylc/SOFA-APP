function getReports() {
  var url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/reports?orderBy=createdAt desc&pageSize=50';

  var token = getAdminToken();
  var response = UrlFetchApp.fetch(url, {
    headers: { 'Authorization': 'Bearer ' + token },
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());
  var reports = [];
  var pendingCount = 0;

  if (data.documents) {
    reports = data.documents.map(function(doc) {
      var fields = doc.fields;
      var id = doc.name.split('/').pop();

      var status = fields.status ? fields.status.stringValue : 'pending';
      if (status === 'pending') pendingCount++;
      return {
        id: id,
        reportedUser: fields.reportedUser ? fields.reportedUser.stringValue : '',
        reportedBy: fields.reportedBy ? fields.reportedBy.stringValue : '',
        productName: fields.productName ? fields.productName.stringValue : '',
        reason: fields.reason ? fields.reason.stringValue : '',
        status: status,
        createdAt: fields.createdAt ? fields.createdAt.timestampValue : '',
      };
    });
  }

  return {
    reports: reports,
    pendingCount: pendingCount,
    totalCount: reports.length,
  };
}

// BAN済みユーザー一覧を取得
function getBannedUsers() {
  var url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/users?pageSize=100';

  var token = getAdminToken();
  var response = UrlFetchApp.fetch(url, {
    headers: { 'Authorization': 'Bearer ' + token },
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());
  var bannedUsers = [];

  if (data.documents) {
    data.documents.forEach(function(doc) {
      var fields = doc.fields;
      if (fields.banned && fields.banned.booleanValue === true) {
        bannedUsers.push({
          uid: doc.name.split('/').pop(),
          email: fields.email ? fields.email.stringValue : '',
          bannedAt: fields.bannedAt ? fields.bannedAt.timestampValue : '',
        });
      }
    });
  }

  return { bannedUsers: bannedUsers };
}

// ユーザーをBAN
function banUser(email, reportId) {
  var token = getAdminToken();

  // メールアドレスからuidを検索
  var searchUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents:runQuery';

  var query = {
    structuredQuery: {
      from: [{ collectionId: 'users' }],
      where: {
        fieldFilter: {

          field: { fieldPath: 'email' },
          op: 'EQUAL',
          value: { stringValue: email }
        }
      },
      limit: 1
    }
  };

  var searchResponse = UrlFetchApp.fetch(searchUrl, {
    method: 'POST',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(query),
    muteHttpExceptions: true
  });

  var searchResult = JSON.parse(searchResponse.getContentText());

  if (!searchResult[0] || !searchResult[0].document) {
    throw new Error('ユーザーが見つかりません');
  }

  var uid = searchResult[0].document.name.split('/').pop();

  // BANフラグを立てる
  var banUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/users/' + uid +
    '?updateMask.fieldPaths=banned&updateMask.fieldPaths=bannedAt';

  UrlFetchApp.fetch(banUrl, {
    method: 'PATCH',

    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify({
      fields: {
        banned: { booleanValue: true },
        bannedAt: { timestampValue: new Date().toISOString() }
      }
    }),
    muteHttpExceptions: true
  });

  // 通報のステータスを resolved に更新
  if (reportId) {
    var reportUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
      '/databases/(default)/documents/reports/' + reportId +
      '?updateMask.fieldPaths=status';

    UrlFetchApp.fetch(reportUrl, {
      method: 'PATCH',
      contentType: 'application/json',
      headers: { 'Authorization': 'Bearer ' + token },
      payload: JSON.stringify({
        fields: { status: { stringValue: 'resolved' } }
      }),
      muteHttpExceptions: true
    });
  }

  return { success: true };
}
