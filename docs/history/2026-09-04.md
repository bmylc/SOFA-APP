// 広告一覧を取得
function getAdvertisers() {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  var url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/advertisers?pageSize=100';

  var response = UrlFetchApp.fetch(url, {
    headers: { 'Authorization': 'Bearer ' + token },
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());
  var advertisers = [];

  if (data.documents) {
    advertisers = data.documents.map(function(doc) {
      var fields = doc.fields;
      var id = doc.name.split('/').pop();
      return {
        id: id,
        name: fields.name ? fields.name.stringValue : '',
        category: fields.category ? fields.category.stringValue : '',
        description: fields.description ? fields.description.stringValue : '',
        imageUrl: fields.imageUrl ? fields.imageUrl.stringValue : '',
        businessHours: fields.businessHours ? fields.businessHours.stringValue : '',
        menu: fields.menu ? fields.menu.arrayValue.values.map(function(v) {
          return v.stringValue;
        }) : [],
        link: fields.link ? fields.link.stringValue : '',

        isActive: fields.isActive ? fields.isActive.booleanValue : true,
      };
    });
  }

  return { advertisers: advertisers };
}

// 広告を追加・更新
function saveAdvertiser(data) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  var url;
  var method;

  if (data.id) {
    // 更新
    url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
      '/databases/(default)/documents/advertisers/' + data.id;
    method = 'PATCH';
  } else {
    // 新規作成
    url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
      '/databases/(default)/documents/advertisers';
    method = 'POST';
  }

  var menuValues = data.menu.map(function(item) {
    return { stringValue: item };
  });

  var payload = {
    fields: {
      name: { stringValue: data.name },
      category: { stringValue: data.category },
      description: { stringValue: data.description },
      imageUrl: { stringValue: data.imageUrl || '' },
      businessHours: { stringValue: data.businessHours || '' },
      menu: { arrayValue: { values: menuValues } },
      link: { stringValue: data.link || '' },
      isActive: { booleanValue: data.isActive !== false },
      updatedAt: { timestampValue: new Date().toISOString() },
    }
  };

  if (!data.id) {
    payload.fields.createdAt = { timestampValue: new Date().toISOString() };
  }

  var response = UrlFetchApp.fetch(url, {
    method: method,
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });

  var result = JSON.parse(response.getContentText());

  if (result.error) {
    return { success: false, message: result.error.message };
  }

  return { success: true };
}

// 広告を削除
function deleteAdvertiser(id) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  var url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/advertisers/' + id;

  UrlFetchApp.fetch(url, {
    method: 'DELETE',
    headers: { 'Authorization': 'Bearer ' + token },
    muteHttpExceptions: true
  });

  return { success: true };
}

// 広告の表示/非表示を切り替え
function toggleAdvertiserActive(id, isActive) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  var url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/advertisers/' + id +
    '?updateMask.fieldPaths=isActive';

  UrlFetchApp.fetch(url, {
    method: 'PATCH',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify({
      fields: { isActive: { booleanValue: isActive } }
    }),
    muteHttpExceptions: true
  });

  return { success: true };
}

ステップ 4：admin.html のJavaScriptに追加

javascript
var CATEGORY_LABELS = {
  coop: '生協',
  kitchencar: 'キッチンカー',
  cafeteria: '食堂',
  bakery: 'ベーカリー',
  circle: '学生団体',
};

// 広告一覧を読み込む
function loadAdvertisers() {
  google.script.run
    .withSuccessHandler(function(data) {
      document.getElementById('advertisersLoading').style.display = 'none';

      var list = document.getElementById('advertisersList');
      if (data.advertisers.length === 0) {
        list.innerHTML = '<div class="empty">広告はまだ登録されていません</div>';
        return;
      }

      list.innerHTML = data.advertisers.map(function(a) {
        return '<div class="advertiser-card">' +
          '<div class="advertiser-card-header">' +
            '<div style="display:flex; gap:12px; align-items:flex-start;">' +
              (a.imageUrl ? '<img src="' + a.imageUrl + '" class="advertiser-image" />' : '') +
              '<div>' +
                '<div class="advertiser-name">' + a.name + '</div>' +
                '<div class="advertiser-category">' + (CATEGORY_LABELS[a.category] || a.category) + '</div>' +
                (a.businessHours ? '<div style="font-size:12px;color:#06534B;margin-top:4px;">🕐 ' + a.businessHours + '</div>' : '') +
              '</div>' +
            '</div>' +
            '<span class="badge ' + (a.isActive ? 'badge-active' : 'badge-inactive') + '">' +

              (a.isActive ? '表示中' : '非表示') +
            '</span>' +
          '</div>' +
          '<div class="actions">' +
            '<button class="btn btn-save" onclick="showAdvertiserForm(' + JSON.stringify(a).replace(/'/g, "\\'") + ')">編集</button>' +
            '<button class="btn btn-dismiss" onclick="toggleActive(\'' + a.id + '\', ' + !a.isActive + ')">' +
              (a.isActive ? '非表示にする' : '表示する') +
            '</button>' +
            '<button class="btn btn-ban" onclick="deleteAdvertiser(\'' + a.id + '\')">削除</button>' +
          '</div>' +
        '</div>';
      }).join('');
    })
    .withFailureHandler(function(error) {
      document.getElementById('advertisersLoading').innerHTML = 'エラー: ' + error.message;
    })
    .getAdvertisers();
}

// フォームを表示
function showAdvertiserForm(advertiser) {
  document.getElementById('advertiserForm').style.display = 'block';

  if (advertiser) {
    document.getElementById('formTitle').textContent = '広告を編集';
    document.getElementById('advertiserId').value = advertiser.id;
    document.getElementById('advertiserName').value = advertiser.name;
    document.getElementById('advertiserCategory').value = advertiser.category;
    document.getElementById('advertiserDescription').value = advertiser.description;
    document.getElementById('advertiserImageUrl').value = advertiser.imageUrl;
    document.getElementById('advertiserBusinessHours').value = advertiser.businessHours;
    document.getElementById('advertiserMenu').value = advertiser.menu.join('\n');

    document.getElementById('advertiserLink').value = advertiser.link;
    document.getElementById('advertiserIsActive').checked = advertiser.isActive;
  } else {
    document.getElementById('formTitle').textContent = '広告を追加';
    document.getElementById('advertiserId').value = '';
    document.getElementById('advertiserName').value = '';
    document.getElementById('advertiserCategory').value = 'coop';
    document.getElementById('advertiserDescription').value = '';
    document.getElementById('advertiserImageUrl').value = '';
    document.getElementById('advertiserBusinessHours').value = '';
    document.getElementById('advertiserMenu').value = '';
    document.getElementById('advertiserLink').value = '';
    document.getElementById('advertiserIsActive').checked = true;
  }
}

// フォームを非表示
function hideAdvertiserForm() {
  document.getElementById('advertiserForm').style.display = 'none';
}

// 広告を保存
function saveAdvertiser() {
  var menuText = document.getElementById('advertiserMenu').value;
  var menu = menuText.split('\n').filter(function(item) { return item.trim() !== ''; });

  var data = {
    id: document.getElementById('advertiserId').value || null,
    name: document.getElementById('advertiserName').value,
    category: document.getElementById('advertiserCategory').value,
    description: document.getElementById('advertiserDescription').value,
    imageUrl: document.getElementById('advertiserImageUrl').value,

    businessHours: document.getElementById('advertiserBusinessHours').value,
    menu: menu,
    link: document.getElementById('advertiserLink').value,
    isActive: document.getElementById('advertiserIsActive').checked,
  };

  if (!data.name) {
    alert('名前を入力してください');
    return;
  }

  google.script.run
    .withSuccessHandler(function(result) {
      if (result.success) {
        hideAdvertiserForm();
        loadAdvertisers();
        alert('保存しました');
      } else {
        alert('エラー: ' + result.message);
      }
    })
    .withFailureHandler(function(error) {
      alert('エラー: ' + error.message);
    })
    .saveAdvertiser(data);
}

// 表示・非表示を切り替え
function toggleActive(id, isActive) {
  google.script.run
    .withSuccessHandler(function() {
      loadAdvertisers();

    })
    .toggleAdvertiserActive(id, isActive);
}

// 広告を削除
function deleteAdvertiser(id) {
  if (!confirm('この広告を削除しますか？')) return;
  google.script.run
    .withSuccessHandler(function() {
      loadAdvertisers();
      alert('削除しました');
    })
    .deleteAdvertiser(id);
}
