function deleteUserProducts(email) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  // 商品を検索
  var searchUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents:runQuery';

  var query = {
    structuredQuery: {
      from: [{ collectionId: 'products' }],
      where: {
        fieldFilter: {
          field: { fieldPath: 'seller' },
          op: 'EQUAL',
          value: { stringValue: email }
        }
      }
    }
  };

  var response = UrlFetchApp.fetch(searchUrl, {
    method: 'POST',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(query),
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());
  var deletedCount = 0;

  data.forEach(function(item) {
    if (!item.document) return;
    var docName = item.document.name;
    var productId = docName.split('/').pop();

    // ← バックアップを banned_products に保存
    var backupUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
      '/databases/(default)/documents/banned_products/' + productId;

    UrlFetchApp.fetch(backupUrl, {
      method: 'PATCH',
      contentType: 'application/json',
      headers: { 'Authorization': 'Bearer ' + token },
      payload: JSON.stringify({
        fields: Object.assign({}, item.document.fields, {
          bannedEmail: { stringValue: email },
          bannedAt: { timestampValue: new Date().toISOString() }
        })
      }),
      muteHttpExceptions: true
    });

    // 元の商品を削除
    var deleteUrl = 'https://firestore.googleapis.com/v1/' + docName;
    UrlFetchApp.fetch(deleteUrl, {
      method: 'DELETE',
      headers: { 'Authorization': 'Bearer ' + token },
      muteHttpExceptions: true
    });

    deletedCount++;

  });

  return { success: true, deletedCount: deletedCount };
}

function unbanUser(uid, email) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  // BANフラグを解除
  var url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/users/' + uid +
    '?updateMask.fieldPaths=banned';

  UrlFetchApp.fetch(url, {
    method: 'PATCH',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify({
      fields: { banned: { booleanValue: false } }
    }),
    muteHttpExceptions: true
  });

  // banned_products からバックアップを取得して復元
  var searchUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents:runQuery';

  var query = {
    structuredQuery: {
      from: [{ collectionId: 'banned_products' }],
      where: {
        fieldFilter: {
          field: { fieldPath: 'bannedEmail' },
          op: 'EQUAL',
          value: { stringValue: email }

        }
      }
    }
  };

  var response = UrlFetchApp.fetch(searchUrl, {
    method: 'POST',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(query),
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());
  var restoredCount = 0;

  data.forEach(function(item) {
    if (!item.document) return;
    var docName = item.document.name;
    var productId = docName.split('/').pop();

    // bannedEmail と bannedAt を除いたフィールドを復元
    var fields = Object.assign({}, item.document.fields);
    delete fields.bannedEmail;
    delete fields.bannedAt;

    // products に復元
    var restoreUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
      '/databases/(default)/documents/products/' + productId;

    UrlFetchApp.fetch(restoreUrl, {
      method: 'PATCH',

      contentType: 'application/json',
      headers: { 'Authorization': 'Bearer ' + token },
      payload: JSON.stringify({ fields: fields }),
      muteHttpExceptions: true
    });

    // banned_products からバックアップを削除
    var deleteUrl = 'https://firestore.googleapis.com/v1/' + docName;
    UrlFetchApp.fetch(deleteUrl, {
      method: 'DELETE',
      headers: { 'Authorization': 'Bearer ' + token },
      muteHttpExceptions: true
    });

    restoredCount++;
  });

  return { success: true, restoredCount: restoredCount };
}

javascript
import { createContext, useContext, useState, useEffect } from 'react';
import { collection, query, where, getDocs } from 'firebase/firestore';
import { db } from '../firebase';

const UserContext = createContext();

export function UserProvider({ children }) {
  const [userCache, setUserCache] = useState({});

  // メールアドレスから氏名を取得（キャッシュ付き）
  const getUserName = async (email) => {
    if (!email || email === 'system') return 'システム';

    // キャッシュに存在すればそれを返す
    if (userCache[email]) return userCache[email];

    try {
      const q = query(collection(db, 'users'), where('email', '==', email));
      const snapshot = await getDocs(q);
      if (!snapshot.empty) {
        const name = snapshot.docs[0].data().name || email;
        setUserCache(prev => ({ ...prev, [email]: name }));
        return name;
      }
    } catch (error) {
      console.error(error);
    }
    return email;
  };

  return (
    <UserContext.Provider value={{ getUserName, userCache }}>

      {children}
    </UserContext.Provider>
  );
}

export function useUsers() {
  return useContext(UserContext);
}

  const [buyerNames, setBuyerNames] = useState({}); // ← 買い手の表示名

  useEffect(() => {
    const q = query(
      collection(db, 'chats'),
      where('productId', '==', productId)
    );
    const unsubscribe = onSnapshot(q, (snapshot) => {
      const data = snapshot.docs.map((doc) => ({
        id: doc.id,
        ...doc.data(),
      }));
      setChats(data);
      setLoading(false);
    });
    return unsubscribe;
  }, [productId]);

  // チャットが更新されたら買い手の名前を取得
  useEffect(() => {
    const fetchNames = async () => {
      const names = {};
