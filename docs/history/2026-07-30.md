javascript
import { useState, useEffect } from 'react';
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { createBottomTabNavigator } from '@react-navigation/bottom-tabs';
import { Text, ActivityIndicator, View } from 'react-native';
import { onAuthStateChanged } from 'firebase/auth';
import { doc, getDoc, onSnapshot } from 'firebase/firestore';
import { auth, db } from './firebase';
import { FavoritesProvider } from './context/FavoritesContext';
import BannedScreen from './screens/BannedScreen';

// ... 既存のインポート ...

export default function App() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [isBanned, setIsBanned] = useState(false);

  useEffect(() => {
    const unsubscribeAuth = onAuthStateChanged(auth, async (user) => {
      if (user) {
        setUser(user);
        setLoading(false);

        // BANステータスをリアルタイムで監視
        const userRef = doc(db, 'users', user.uid);
        const unsubscribeUser = onSnapshot(userRef, (snap) => {
          if (snap.exists()) {
            setIsBanned(snap.data().banned === true);
          }
        });

        return unsubscribeUser;
      } else {
        setUser(null);
        setIsBanned(false);
        setLoading(false);
      }
    });

    return unsubscribeAuth;
  }, []);

  if (loading) {
    return (
      <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center' }}>
        <ActivityIndicator size="large" color="#06534B" />
      </View>
    );
  }

  return (
    <FavoritesProvider>
      <NavigationContainer>
        {!user ? (
          <AuthStack />
        ) : isBanned ? (
          <BannedScreen /> // ← BANされていたら専用画面
        ) : (
          <MainScreen />
        )}
      </NavigationContainer>
    </FavoritesProvider>
  );

}
変更のポイント

変更前：
  ログイン時に一度だけBANチェック
  → ログイン後にBANしても効かない

変更後：
  onSnapshot でBANステータスをリアルタイム監視
  → ログイン中にBANすると即座に専用画面に遷移
  → アプリを再起動しなくても反映される

// BANされたユーザーの商品を削除
function deleteUserProducts(email) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  // メールアドレスで商品を検索
  var searchUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents:runQuery';

  var query = {
    structuredQuery: {
      from: [{ collectionId: 'products' }],
      where: {
        fieldFilter: {
          field: { fieldPath: 'seller' },
          op: 'EQUAL',
          value: { stringValue: email }
        }
      }
    }
  };

  var response = UrlFetchApp.fetch(searchUrl, {
    method: 'POST',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(query),
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());

  var deletedCount = 0;

  data.forEach(function(item) {
    if (!item.document) return;
    var docName = item.document.name;
    var deleteUrl = 'https://firestore.googleapis.com/v1/' + docName;

    UrlFetchApp.fetch(deleteUrl, {
      method: 'DELETE',
      headers: { 'Authorization': 'Bearer ' + token },
      muteHttpExceptions: true
    });
    deletedCount++;
  });

  return { success: true, deletedCount: deletedCount };
}

function banUser(email, reportId) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  // ユーザーを検索
  var searchUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents:runQuery';

  var query = {
    structuredQuery: {
      from: [{ collectionId: 'users' }],
      where: {
        fieldFilter: {
          field: { fieldPath: 'email' },
          op: 'EQUAL',
          value: { stringValue: email }
        }
      },
      limit: 1
    }
  };

  var searchResponse = UrlFetchApp.fetch(searchUrl, {
    method: 'POST',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(query),
    muteHttpExceptions: true
  });

  var searchResult = JSON.parse(searchResponse.getContentText());

  if (!searchResult[0] || !searchResult[0].document) {
    throw new Error('ユーザーが見つかりません');
  }

  var uid = searchResult[0].document.name.split('/').pop();

  // BANフラグを立てる
  var banUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/users/' + uid +
    '?updateMask.fieldPaths=banned&updateMask.fieldPaths=bannedAt';

  UrlFetchApp.fetch(banUrl, {
    method: 'PATCH',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify({
      fields: {
        banned: { booleanValue: true },
        bannedAt: { timestampValue: new Date().toISOString() }
      }
    }),
    muteHttpExceptions: true
  });

  // ← 商品を削除
  var deleteResult = deleteUserProducts(email);
  console.log('削除した商品数:', deleteResult.deletedCount);

  // 通報のステータスを resolved に更新
  if (reportId) {
    var reportUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
      '/databases/(default)/documents/reports/' + reportId +

      '?updateMask.fieldPaths=status';

    UrlFetchApp.fetch(reportUrl, {
      method: 'PATCH',
      contentType: 'application/json',
      headers: { 'Authorization': 'Bearer ' + token },
      payload: JSON.stringify({
        fields: { status: { stringValue: 'resolved' } }
      }),
      muteHttpExceptions: true
    });
  }

  return { success: true, deletedCount: deleteResult.deletedCount };
}
