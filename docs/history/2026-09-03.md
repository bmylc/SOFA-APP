function verifyAndIssueToken(uid) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');

  // cardUid でFirestoreを検索
  var searchUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents:runQuery';

  var query = {
    structuredQuery: {
      from: [{ collectionId: 'users' }],
      where: {
        fieldFilter: {
          field: { fieldPath: 'cardUid' },
          op: 'EQUAL',
          value: { stringValue: uid }
        }
      },
      limit: 1
    }
  };

  var response = UrlFetchApp.fetch(searchUrl, {
    method: 'POST',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(query),
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());

  // 紐づいたアカウントが見つからない場合
  if (!data[0] || !data[0].document) {

export function generateTradeNumber() {
  const now = new Date();
  const year = now.getFullYear();
  const month = String(now.getMonth() + 1).padStart(2, '0');
  const day = String(now.getDate()).padStart(2, '0');
  const random = String(Math.floor(Math.random() * 9999)).padStart(4, '0');
  return `SOFA-${year}${month}${day}-${random}`;
}

export default function MyPageScreen({ navigation, route }) {
  const returnTo = route.params?.returnTo || null;

  // ヘッダーの戻るボタンを設定
  useEffect(() => {
    navigation.setOptions({
      headerLeft: () => (
        <TouchableOpacity
          onPress={() => {
            if (returnTo === 'Campus') {

          </Text>
        </TouchableOpacity>
      ),
    });
