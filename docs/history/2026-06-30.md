GAS側の registerUser 関数を作成します。Firebaseにユーザーを登録する処理ですね。


javascript
function doGet() {
  return HtmlService.createHtmlOutputFromFile('register')
    .setTitle('会員登録')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function registerUser(data) {
  try {
    // Firebase Authentication REST API でユーザー登録
    var apiKey = 'ここにFirebaseのAPIキーを入力';
    
    var authUrl = 'https://identitytoolkit.googleapis.com/v1/accounts:signUp?key=' + apiKey;
    
    var authPayload = JSON.stringify({
      email: data.email,
      password: data.password,
      returnSecureToken: true
    });
    
    var authResponse = UrlFetchApp.fetch(authUrl, {
      method: 'POST',
      contentType: 'application/json',
      payload: authPayload,
      muteHttpExceptions: true
    });
    
    var authResult = JSON.parse(authResponse.getContentText());
    
    // エラーチェック
    if (authResult.error) {
      var errorCode = authResult.error.message;
      if (errorCode === 'EMAIL_EXISTS') {

        return { success: false, message: 'このメールアドレスはすでに登録されています' };
      }
      if (errorCode === 'WEAK_PASSWORD : Password should be at least 6 characters') {
        return { success: false, message: 'パスワードは6文字以上にしてください' };
      }
      return { success: false, message: '登録に失敗しました: ' + errorCode };
    }
    
    var uid = authResult.localId;
    
    // Firestore にユーザー情報を保存
    var projectId = 'ここにFirebaseのプロジェクトIDを入力';
    var idToken = authResult.idToken;
    
    var firestoreUrl = 'https://firestore.googleapis.com/v1/projects/' + projectId + '/databases/(default)/documents/users/' + uid;
    
    var firestorePayload = JSON.stringify({
      fields: {
        name: { stringValue: data.name },
        email: { stringValue: data.email },
        affiliation: { stringValue: data.affiliation },
        faculty: { stringValue: data.faculty || '' },
        department: { stringValue: data.department || '' },
        createdAt: { timestampValue: new Date().toISOString() }
      }
    });
    
    var firestoreResponse = UrlFetchApp.fetch(firestoreUrl, {
      method: 'PATCH',
      contentType: 'application/json',
      headers: {
        'Authorization': 'Bearer ' + idToken

      },
      payload: firestorePayload,
      muteHttpExceptions: true
    });
    
    var firestoreResult = JSON.parse(firestoreResponse.getContentText());
    
    if (firestoreResult.error) {
      return { success: false, message: 'ユーザー情報の保存に失敗しました' };
    }
    
    return { success: true };
    
  } catch (error) {
    return { success: false, message: 'エラーが発生しました: ' + error.message };
  }
}
