function unlinkNfc(email) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  // メールアドレスでユーザーを検索
  var searchUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents:runQuery';

  var query = {
    structuredQuery: {
      from: [{ collectionId: 'users' }],
      where: {
        fieldFilter: {
          field: { fieldPath: 'email' },
          op: 'EQUAL',
          value: { stringValue: email }
        }
      },
      limit: 1
    }
  };

  var response = UrlFetchApp.fetch(searchUrl, {
    method: 'POST',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(query),
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());

  if (!data[0] || !data[0].document) {
    return ContentService
      .createTextOutput(JSON.stringify({
        success: false,
        message: 'ユーザーが見つかりません'
      }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  var docName = data[0].document.name;

  // cardUid を削除
  var updateUrl = 'https://firestore.googleapis.com/v1/' + docName +
    '?updateMask.fieldPaths=cardUid';

  UrlFetchApp.fetch(updateUrl, {
    method: 'PATCH',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify({
      fields: {
        cardUid: { nullValue: null }, // ← nullに設定
      }
    }),
    muteHttpExceptions: true
  });

  return ContentService
    .createTextOutput(JSON.stringify({ success: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
