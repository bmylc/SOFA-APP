`````text
// Firebaseカスタムトークンを生成
function createCustomToken(uid) {
  var json = JSON.parse(PropertiesService.getScriptProperties().getProperty('SERVICE_ACCOUNT_JSON'));
  var clientEmail = json.client_email;
  var privateKey = json.private_key;
  var now = Math.floor(Date.now() / 1000);

  var header = Utilities.base64EncodeWebSafe(JSON.stringify({
    alg: 'RS256',
    typ: 'JWT'
  }));

  var claim = Utilities.base64EncodeWebSafe(JSON.stringify({
    iss: clientEmail,
    sub: clientEmail,
    aud: 'https://identitytoolkit.googleapis.com/google.identity.identitytoolkit.v1.IdentityToolkit',
    iat: now,
    exp: now + 3600,
    uid: 'student_' + uid, // ← UIDをFirebaseのユーザーIDとして使用
    claims: {
      isEnrolled: true,
    }
  }));

  var signature = Utilities.base64EncodeWebSafe(
    Utilities.computeRsaSha256Signature(header + '.' + claim, privateKey)
  );

  return header + '.' + claim + '.' + signature;
}

    '/databases/(default)/documents/users/' + firebaseUid;

  var response = UrlFetchApp.fetch(url, {
    headers: { 'Authorization': 'Bearer ' + token },
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());
  return !data.error;
}

import NfcManager, { NfcTech } from 'react-native-nfc-manager';

const GAS_URL = 'あなたのGASのURL';

// NFC初期化
export async function initNfc() {
  return await NfcManager.start();
}

// NFCが使えるか確認
export async function isNfcSupported() {
  return await NfcManager.isSupported();
}

// FeliCaからUIDを読み取る
export async function readStudentCardUid() {
  try {
    await NfcManager.requestTechnology(NfcTech.NfcF);
    const tag = await NfcManager.getTag();
    return tag.id; // UID
  } catch (error) {
    throw error;
  } finally {
    NfcManager.cancelTechnologyRequest();
  }
}

// UIDをGASに送ってカスタムトークンを取得
export async function verifyStudentCard(uid) {
  try {
    const response = await fetch(
      GAS_URL + '?action=verify&uid=' + uid

    );
    const data = await response.json();
    return data;
  } catch (error) {
    throw new Error('サーバーとの通信に失敗しました');
  }
}

// ユーザーが既に存在するか確認
export async function checkUserExists(firebaseUid) {
  try {
    const response = await fetch(
      GAS_URL + '?action=checkUser&firebaseUid=' + firebaseUid
    );
    const data = await response.json();
    return data.exists;
  } catch (error) {
    return false;
  }
}

// 在校生判定（Firestoreのuserデータから）
export function isEnrolledStudent(userDoc) {
  return userDoc.isEnrolled === true;
}

import { useState, useEffect } from 'react';
import { View, Text, TouchableOpacity, StyleSheet, ActivityIndicator, Alert } from 'react-native';
import { signInWithCustomToken } from 'firebase/auth';
import { auth } from '../firebase';
import {
  initNfc,
  isNfcSupported,
  readStudentCardUid,
  verifyStudentCard,
  checkUserExists,
} from '../utils/checkEnrollment';

export default function NfcLoginScreen({ navigation }) {
  const [loading, setLoading] = useState(false);
  const [status, setStatus] = useState('idle'); // idle / reading / verifying / success / error
  const [nfcSupported, setNfcSupported] = useState(true);

  useEffect(() => {
    const checkNfc = async () => {
      const supported = await isNfcSupported();
      setNfcSupported(supported);
      if (supported) await initNfc();
    };
    checkNfc();
  }, []);

  const handleNfcLogin = async () => {
    setLoading(true);
    setStatus('reading');

    try {
      // Step1: NFCで学生証を読み取る

      const uid = await readStudentCardUid();
      setStatus('verifying');

      // Step2: GASで在校生確認＆カスタムトークン取得
      const result = await verifyStudentCard(uid);

      if (!result.success) {
        Alert.alert('認証失敗', result.message || '在校生として確認できませんでした');
        setStatus('error');
        setLoading(false);
        return;
      }

      // Step3: カスタムトークンでFirebaseにログイン
      const userCredential = await signInWithCustomToken(auth, result.token);
      const firebaseUid = userCredential.user.uid;

      // Step4: アカウントが存在するか確認
      const exists = await checkUserExists(firebaseUid);

      if (!exists) {
        // 初回ログイン → プロフィール入力画面へ
        setStatus('success');
        navigation.navigate('NfcProfile', { firebaseUid, cardUid: uid });
      } else {
        // 2回目以降 → そのままログイン
        setStatus('success');
      }
    } catch (error) {
      console.error(error);
      Alert.alert('エラー', 'NFC読み取りに失敗しました。もう一度試してください。');
      setStatus('error');

    } finally {
      setLoading(false);
    }
  };

  const getStatusMessage = () => {
    switch (status) {
      case 'reading': return '学生証を読み取り中...';
      case 'verifying': return 'サーバーで確認中...';
      case 'success': return '認証成功！';
      case 'error': return 'もう一度お試しください';
      default: return 'スマホに学生証をかざしてください';
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>学生証でログイン</Text>

      {!nfcSupported ? (
        <View style={styles.errorBox}>
          <Text style={styles.errorText}>
            このデバイスはNFCに対応していません
          </Text>
        </View>
      ) : (
        <>
          <TouchableOpacity
            style={[styles.nfcButton, loading && styles.nfcButtonLoading]}
            onPress={handleNfcLogin}
            disabled={loading}
          >

            {loading ? (
              <ActivityIndicator size="large" color="#fff" />
            ) : (
              <>
                <Text style={styles.nfcIcon}>🎓</Text>
                <Text style={styles.nfcButtonText}>学生証をかざす</Text>
              </>
            )}
          </TouchableOpacity>

          <Text style={styles.statusText}>{getStatusMessage()}</Text>

          {/* アニメーション用の波紋 */}
          {loading && (
            <View style={styles.ripple} />
          )}
        </>
      )}
    </View>
  );
}

const FACULTIES = ['文学部', '法学部', '経済学部', '理工学部', '医学部', '工学部', '教育学部', 'その他'];

        name: name.trim(),

        faculty,
        department,
        cardUid,
        isEnrolled: true,

    } catch (error) {
      Alert.alert('エラー', 'プロフィールの保存に失敗しました');

      setLoading(false);
    }
  };

  return (
    <ScrollView contentContainerStyle={styles.container}>
`````

## 6. Claude

`````text

Claudeが返答しました: 良い質問です！理由はセキュリティです。

良い質問です！理由はセキュリティです。

GASを経由する理由

Firebaseのカスタムトークン生成には秘密鍵が必要です。


カスタムトークン生成に必要なもの：
・サービスアカウントの秘密鍵
・サービスアカウントのメールアドレス

この秘密鍵をアプリ内に直接書くと危険です。


❌ アプリに秘密鍵を直接書く
　→ APKを解析されると秘密鍵が漏洩
　→ 誰でもカスタムトークンを発行できてしまう
　→ なりすましログインが可能になる

✅ GASサーバー上に秘密鍵を置く
　→ アプリには秘密鍵が存在しない
　→ サーバー側でUID照合してからトークン発行
`````

`````text
function getAdminToken() {
  var props = PropertiesService.getScriptProperties();
  var json = JSON.parse(props.getProperty('SERVICE_ACCOUNT_JSON'));
  var clientEmail = json.client_email;
  var privateKey = json.private_key;

  var now = Math.floor(Date.now() / 1000);
  var header = Utilities.base64EncodeWebSafe(JSON.stringify({
    alg: 'RS256', typ: 'JWT'
  }));

  var claim = Utilities.base64EncodeWebSafe(JSON.stringify({
    iss: clientEmail,
    sub: clientEmail,
    aud: 'https://oauth2.googleapis.com/token',
    iat: now,
    exp: now + 3600,
    scope: 'https://www.googleapis.com/auth/datastore'
  }));

  var signature = Utilities.base64EncodeWebSafe(
    Utilities.computeRsaSha256Signature(header + '.' + claim, privateKey)
  );

  var jwt = header + '.' + claim + '.' + signature;

  var response = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', {
    method: 'POST',
    contentType: 'application/x-www-form-urlencoded',
    payload: 'grant_type=urn:ietf:params:oauth:grant-type:jwt-bearer&assertion=' + jwt,
`````

`````text
        const userDoc = await getDoc(doc(db, 'users', auth.currentUser.uid));

      />

      <Text style={styles.sectionTitle}>所属 *</Text>
`````
`````text
      <View style={styles.pickerContainer}>
`````
