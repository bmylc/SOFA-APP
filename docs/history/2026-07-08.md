function getReports() {
  var props = PropertiesService.getScriptProperties();

  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  var url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents:runQuery';

  var query = {
    structuredQuery: {
      from: [{ collectionId: 'reports' }],
      where: {
        fieldFilter: {
          field: { fieldPath: 'status' },
          op: 'EQUAL',
          value: { stringValue: 'pending' }
        }
      },
      orderBy: [{ field: { fieldPath: 'createdAt' }, direction: 'DESCENDING' }]
    }
  };

  var response = UrlFetchApp.fetch(url, {
    method: 'POST',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(query),
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());
  var reports = [];

  data.forEach(function(item) {

    if (!item.document) return;
    var fields = item.document.fields;
    var id = item.document.name.split('/').pop();
    reports.push({
      id: id,
      reportedUser: fields.reportedUser ? fields.reportedUser.stringValue : '',
      reportedBy: fields.reportedBy ? fields.reportedBy.stringValue : '',
      productName: fields.productName ? fields.productName.stringValue : '',
      reason: fields.reason ? fields.reason.stringValue : '',
      status: fields.status ? fields.status.stringValue : 'pending',
      createdAt: fields.createdAt ? fields.createdAt.timestampValue : '',
    });
  });

  return { reports: reports, totalCount: reports.length };
}

function getUserReports() {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  var url = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/reports?pageSize=200';

  var response = UrlFetchApp.fetch(url, {
    headers: { 'Authorization': 'Bearer ' + token },
    muteHttpExceptions: true
  });

  var data = JSON.parse(response.getContentText());
  var userMap = {};

  if (data.documents) {
    data.documents.forEach(function(doc) {
      var fields = doc.fields;
      var email = fields.reportedUser ? fields.reportedUser.stringValue : '';
      var status = fields.status ? fields.status.stringValue : 'pending';
      var createdAt = fields.createdAt ? fields.createdAt.timestampValue : '';

      if (!userMap[email]) {
        userMap[email] = {
          email: email,
          totalCount: 0,
          pendingCount: 0,
          resolvedCount: 0,
          reviewedCount: 0,
          latestReport: '',
          banned: false,
        };
      }

      userMap[email].totalCount++;
      if (status === 'pending') userMap[email].pendingCount++;
      if (status === 'resolved') userMap[email].resolvedCount++;
      if (status === 'reviewed') userMap[email].reviewedCount++;
      if (!userMap[email].latestReport || createdAt > userMap[email].latestReport) {
        userMap[email].latestReport = createdAt;
      }
    });
  }

  // BANステータスを確認
  var usersUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/users?pageSize=100';

  var usersResponse = UrlFetchApp.fetch(usersUrl, {
    headers: { 'Authorization': 'Bearer ' + token },
    muteHttpExceptions: true
  });

  var usersData = JSON.parse(usersResponse.getContentText());
  if (usersData.documents) {
    usersData.documents.forEach(function(doc) {
      var fields = doc.fields;
      var email = fields.email ? fields.email.stringValue : '';
      var banned = fields.banned ? fields.banned.booleanValue : false;
      var uid = doc.name.split('/').pop();
      if (userMap[email]) {
        userMap[email].banned = banned;
        userMap[email].uid = uid;
      }
    });
  }

  // 通報回数が多い順にソート
  var users = Object.values(userMap).sort(function(a, b) {
    return b.totalCount - a.totalCount;
  });

  return { users: users };
}

function banUser(email, reportId) {
  var props = PropertiesService.getScriptProperties();
  var PROJECT_ID = props.getProperty('FIREBASE_PROJECT_ID');
  var token = getAdminToken();

  var searchUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents:runQuery';

  var query = {

    structuredQuery: {
      from: [{ collectionId: 'users' }],
      where: {
        fieldFilter: {
          field: { fieldPath: 'email' },
          op: 'EQUAL',
          value: { stringValue: email }
        }
      },
      limit: 1
    }
  };

  var searchResponse = UrlFetchApp.fetch(searchUrl, {
    method: 'POST',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify(query),
    muteHttpExceptions: true
  });

  var searchResult = JSON.parse(searchResponse.getContentText());
  if (!searchResult[0] || !searchResult[0].document) {
    throw new Error('ユーザーが見つかりません');
  }

  var uid = searchResult[0].document.name.split('/').pop();

  var banUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
    '/databases/(default)/documents/users/' + uid +
    '?updateMask.fieldPaths=banned&updateMask.fieldPaths=bannedAt';

  UrlFetchApp.fetch(banUrl, {
    method: 'PATCH',
    contentType: 'application/json',
    headers: { 'Authorization': 'Bearer ' + token },
    payload: JSON.stringify({
      fields: {
        banned: { booleanValue: true },
        bannedAt: { timestampValue: new Date().toISOString() }
      }
    }),
    muteHttpExceptions: true
  });

  if (reportId) {
    var reportUrl = 'https://firestore.googleapis.com/v1/projects/' + PROJECT_ID +
      '/databases/(default)/documents/reports/' + reportId +
      '?updateMask.fieldPaths=status';

    UrlFetchApp.fetch(reportUrl, {
      method: 'PATCH',
      contentType: 'application/json',
      headers: { 'Authorization': 'Bearer ' + token },
      payload: JSON.stringify({
        fields: { status: { stringValue: 'resolved' } }
      }),
      muteHttpExceptions: true
    });
  }

  return { success: true };
}

function getAdminToken() {
  var props = PropertiesService.getScriptProperties();
  var json = JSON.parse(props.getProperty('SERVICE_ACCOUNT_JSON'));
  var clientEmail = json.client_email;
  var privateKey = json.private_key;

  var now = Math.floor(Date.now() / 1000);
  var header = Utilities.base64EncodeWebSafe(JSON.stringify({
    alg: 'RS256', typ: 'JWT'
  }));

  var claim = Utilities.base64EncodeWebSafe(JSON.stringify({
    iss: clientEmail,
    sub: clientEmail,
    aud: 'https://oauth2.googleapis.com/token',
    iat: now,
    exp: now + 3600,
    scope: 'https://www.googleapis.com/auth/datastore'
  }));

  var signature = Utilities.base64EncodeWebSafe(
    Utilities.computeRsaSha256Signature(header + '.' + claim, privateKey)
  );

  var jwt = header + '.' + claim + '.' + signature;

  var response = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', {
    method: 'POST',
    contentType: 'application/x-www-form-urlencoded',
    payload: 'grant_type=urn:ietf:params:oauth:grant-type:jwt-bearer&assertion=' + jwt,
    muteHttpExceptions: true
  });

  return JSON.parse(response.getContentText()).access_token;
}

function loadReports() {
  document.getElementById('pendingLoading').style.display = 'block';
  document.getElementById('resolvedLoading').style.display = 'block';

  google.script.run
    .withSuccessHandler(function(data) {
      allReports = data.reports;

      // JavaScript側で分類
      var pending = allReports.filter(function(r) { return r.status === 'pending'; });
      var resolved = allReports.filter(function(r) { return r.status !== 'pending'; });
      var banCount = allReports.filter(function(r) { return r.status === 'resolved'; }).length;
      var ignoreCount = allReports.filter(function(r) { return r.status === 'reviewed'; }).length;

      // 統計を更新
      document.getElementById('statPending').textContent = pending.length;
      document.getElementById('statResolved').textContent = banCount;
      document.getElementById('statDismissed').textContent = ignoreCount;

      // ローディングを非表示
      document.getElementById('pendingLoading').style.display = 'none';

      document.getElementById('resolvedLoading').style.display = 'none';

      // 各タブに描画
      renderPending(pending);
      renderResolved(currentSubTab);
    })
    .withFailureHandler(function(error) {
      document.getElementById('pendingLoading').innerHTML = 'エラー: ' + error.message;
    })
    .getAllReports(); // ← 1つの関数に統合
}

function renderResolved(type) {
  currentSubTab = type;
  var resolved = allReports.filter(function(r) { return r.status !== 'pending'; });

  var list = document.getElementById('resolvedList');
  if (filtered.length === 0) {
    list.innerHTML = '<div class="empty">該当する通報はありません</div>';

function loadUsers() {
  google.script.run
    .withSuccessHandler(function(data) {
      document.getElementById('usersLoading').style.display = 'none';

      // allReports からユーザー単位に集計
      var userMap = {};
      allReports.forEach(function(r) {
        var email = r.reportedUser;
        if (!userMap[email]) {
          userMap[email] = {
            email: email,
            totalCount: 0,
            pendingCount: 0,

            resolvedCount: 0,
            reviewedCount: 0,
            banned: false,
            uid: '',
          };
        }
        userMap[email].totalCount++;
        if (r.status === 'pending') userMap[email].pendingCount++;
        if (r.status === 'resolved') userMap[email].resolvedCount++;
        if (r.status === 'reviewed') userMap[email].reviewedCount++;
      });

      // BANステータスをマージ
      data.bannedUsers.forEach(function(u) {
        if (userMap[u.email]) {
          userMap[u.email].banned = u.banned;
          userMap[u.email].uid = u.uid;
        }
      });

      // 通報回数が多い順にソート
      var users = Object.values(userMap).sort(function(a, b) {
        return b.totalCount - a.totalCount;
      });

      renderUserReports(users);
    })
    .getBannedUsers(); // ← BANステータスだけGASから取得
}

import priceLimits from '../data/priceLimits.json';

// カテゴリごとの上限価格を取得
export function getPriceLimit(category) {
  return priceLimits.categories[category] || priceLimits.default;
}

// 価格バリデーション（上限チェック付き）
export function validatePrice(price, category) {
  if (!price || price.trim() === '') return '価格を入力してください';
  if (isNaN(price)) return '価格は数字で入力してください';
  if (parseInt(price) <= 0) return '価格は1円以上にしてください';

  const limit = getPriceLimit(category);
  if (parseInt(price) > limit) {
    return `${category || 'この商品'}の出品上限は¥${limit.toLocaleString()}です`;
  }

  return null;
}
